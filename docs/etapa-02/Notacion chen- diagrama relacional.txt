1. LEYENDA DE SIMBOLOGÍA (NOTACIÓN CHEN)
--------------------------------------------------------------------------------
- ENTIDAD:                [ NOMBRE_ENTIDAD ]
- ENTIDAD DÉBIL:          [[ NOMBRE_ENTIDAD ]]
- ATRIBUTO:               ( atributo )
- ATRIBUTO CLAVE (PK):    ( *atributo_clave* )
- ATRIBUTO DERIVADO:      ( (atributo_derivado) )
- RELACIÓN:               < NOMBRE_RELACION >
- ATRIBUTO DE RELACIÓN:   < NOMBRE_RELACION > -- ( atributo )
- CARDINALIDAD:           (1:1), (1:N), (N:1), (M:N)


--------------------------------------------------------------------------------
2. ENTIDADES Y ATRIBUTOS
--------------------------------------------------------------------------------

[ PERSONA ] (Entidad Padre)
    ├── ( *dni* )
    ├── ( nombre )
    ├── ( apellido )
    ├── ( cuil )
    ├── ( telefono )
    └── ( email )

[ CLIENTE ] (Especialización de PERSONA)
    ├── ( *dni_cliente* )
    └── ( estado )

[ VENDEDOR ] (Especialización de PERSONA)
    └── ( *dni_vendedor* )

[[ DIRECCION_CLIENTE ]] (Entidad Débil)
    ├── ( calle )
    ├── ( numero )
    ├── ( ciudad )
    ├── ( provincia )
    └── ( codigo_postal )

[ PROVEEDOR ]
    ├── ( *cuit_proveedor* )
    ├── ( razon_social )
    ├── ( email )
    └── ( telefono )

[[ DIRECCION_PROVEEDOR ]] (Entidad Débil)
    ├── ( calle )
    ├── ( numero )
    ├── ( ciudad )
    ├── ( provincia )
    └── ( codigo_postal )

[ MODELO_MOTOCICLETA ]
    ├── ( *id_modelo* )
    ├── ( nombre_modelo )
    ├── ( marca )
    ├── ( cilindrada )
    ├── ( color )
    ├── ( anio )
    ├── ( precio_lista )
    └── ( stock_disponible )

[ MOTOCICLETA ]
    ├── ( *numero_chasis* )
    └── ( numero_motor )

[ ACCESORIO ]
    ├── ( *id_accesorio* )
    ├── ( descripcion )
    ├── ( stock )
    ├── ( categoria )
    ├── ( precio_lista )
    └── ( precio_costo )

[ VENTA ]
    ├── ( *cod_venta* )
    ├── ( fecha )
    ├── ( metodo_pago )
    ├── ( precio_venta_pactado )
    ├── ( estado )
    └── ( monto_total )

[ COTIZACION ]
    ├── ( *id_cotizacion* )
    ├── ( fecha_emision )
    ├── ( fecha_vencimiento )
    └── ( precio_cotizado )

[ HISTORIAL_DE_PRECIO ]
    ├── ( *id_historial* )
    ├── ( precio_nuevo )
    ├── ( precio_anterior )
    └── ( fecha_modificacion )


--------------------------------------------------------------------------------
3. RELACIONES Y CARDINALIDADES (ROMBOS)
--------------------------------------------------------------------------------

-- Jerarquía de Herencia --
[ PERSONA ] ────── IS_A ──────► [ CLIENTE ]
[ PERSONA ] ────── IS_A ──────► [ VENDEDOR ]

-- Relaciones de Clientes y Vendedores --
[ CLIENTE ] (1) ────── < posee_dir_c > ────── (1) [[ DIRECCION_CLIENTE ]]
[ CLIENTE ] (1) ────── < solicita > ───────── (N) [ COTIZACION ]
[ CLIENTE ] (1) ────── < realiza > ────────── (N) [ VENTA ]
[ VENDEDOR ] (1) ───── < atiende > ────────── (N) [ VENTA ]

-- Relaciones de Proveedores --
[ PROVEEDOR ] (1) ──── < posee_dir_p > ────── (1) [[ DIRECCION_PROVEEDOR ]]
[ PROVEEDOR ] (M) ──── < suministra_m > ───── (N) [ MODELO_MOTOCICLETA ]
[ PROVEEDOR ] (M) ──── < suministra_a > ───── (N) [ ACCESORIO ]

-- Relaciones de Modelos e Inventario --
[ MODELO_MOTOCICLETA ] (1) ── < compone > ──── (N) [ MOTOCICLETA ]
[ MODELO_MOTOCICLETA ] (1) ── < registra > ─── (N) [ HISTORIAL_DE_PRECIO ]
[ MODELO_MOTOCICLETA ] (1) ── < cotiza > ───── (N) [ COTIZACION ]

-- Relaciones Transaccionales (Ventas y Detalles) --
[ MOTOCICLETA ] (1) ── < incluida_en > ────── (1) [ VENTA ]

[ VENTA ] (M) ──────── < incluye_acc > ────── (N) [ ACCESORIO ]
                             │
                             ├── ( cantidad )
                             └── ( precio_pactado )
